      def _read_hgc_dist(self):
          """璇诲彇婵€鍏変紶鎰熷櫒鍘熷鍊?(MainWindow Helper)"""
          if not hasattr(self, 'machine') or not self.machi
ne.base_url or not self.machine.connected:
              return None
              
          try:
              url = f"{self.machine.base_url}/printer/objec
ts/query"
              params = {"temperature_sensor hgc_dist": ""}
              # Use a short timeout
              r = self.machine.session.get(url, params=para
ms, headers=self.machine._headers(), timeout=1)
              r.raise_for_status()
              data = r.json()
              
              status = data.get('result', {}).get('status',
 {})
              sensor = status.get('temperature_sensor hgc_d
ist', {})
              val = sensor.get('temperature')
              
              if val is not None:
                  return float(val)
          except Exception:
              pass
          return None
